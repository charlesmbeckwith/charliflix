cmake_minimum_required(VERSION 3.22.1)
project(charliflix LANGUAGES CXX)

get_filename_component(APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(NODE_DIR "${APP_DIR}/libnode")
if(NOT EXISTS "${NODE_DIR}/bin/${ANDROID_ABI}/libnode.so" OR NOT EXISTS "${NODE_DIR}/include/node/node.h")
    message(FATAL_ERROR "Fetch the pinned Android Node.js runtime before building.")
endif()
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_DIR}/bin/${ANDROID_ABI}/libnode.so")
add_library(charliflix SHARED native-lib.cpp)
target_include_directories(charliflix PRIVATE "${NODE_DIR}/include/node")
target_link_libraries(charliflix node log)
target_link_options(charliflix PRIVATE "-Wl,-z,max-page-size=16384")
